// Exemplo da lógica corrigida para encerramento de alarmes e cálculo de alvos function gerenciarAlarme(deveTocar, motivo = "saude") { if (deveTocar) { if (motivo === "crash") { crashAlertaAtivo = true; document.body.classList.add('flash-crash-ativo'); } else { document.body.classList.add('alerta-visual-ativo'); } if (audioPermitido && !alarmeIntervalo) { tocarBip(); let intervaloTempo = crashAlertaAtivo ? 400 : (fatorSaudeAtual <= 1.20 ? 600 : 1200); alarmeIntervalo = setInterval(tocarBip, intervaloTempo); } } else { if (motivo === "crash") { crashAlertaAtivo = false; document.body.classList.remove('flash-crash-ativo'); } if (!crashAlertaAtivo && fatorSaudeAtual > 1.40) { document.body.classList.remove('alerta-visual-ativo'); if (alarmeIntervalo) { clearInterval(alarmeIntervalo); alarmeIntervalo = null; } } } }
